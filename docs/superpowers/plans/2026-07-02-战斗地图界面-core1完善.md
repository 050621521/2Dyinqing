```bash
node <<'NODE'
const fs = require('fs');
const path = 'MyGame/Levels/战斗地图界面.level';
const doc = JSON.parse(fs.readFileSync(path, 'utf8'));
doc.objects = Array.isArray(doc.objects) ? doc.objects : [];

function baseActor(id, name, bpClass, x, y, w, h, color, order, tag = '战斗地图') {
  return {
    active: true,
    animAsset: '',
    animAutoPlay: true,
    animDefaultClip: '',
    bpClass,
    cameraBackground: '#ff1e1e1e',
    cameraIsMain: false,
    cameraOrthographic: true,
    cameraResH: 1080,
    cameraResW: 1920,
    cameraSize: 540,
    colliderEnabled: false,
    colliderH: h,
    colliderOffsetX: 0,
    colliderOffsetY: 0,
    colliderResponse: '阻挡',
    colliderTargets: '',
    colliderW: w,
    components: bpClass === 'builtin/Sprite' ? ['变换', '精灵渲染器'] : ['变换'],
    confinerActor: '',
    confinerEnabled: false,
    confinerMaxX: 500,
    confinerMaxY: 500,
    confinerMinX: -500,
    confinerMinY: -500,
    drawMode: '简单',
    flipX: false,
    flipY: false,
    followLerpSpeed: 5,
    followOffsetX: 0,
    followOffsetY: 0,
    followTarget: '',
    id,
    layer: '默认',
    name,
    orderInLayer: order,
    position: { x, y },
    rotation: 0,
    scale: { x: w / 100, y: h / 100 },
    sortingLayer: '默认',
    spriteColor: color,
    spritePath: '',
    spriteVisible: true,
    static: false,
    tag
  };
}

function upsert(actor) {
  const index = doc.objects.findIndex(o => o.id === actor.id || o.name === actor.name);
  if (index >= 0) doc.objects[index] = { ...doc.objects[index], ...actor };
  else doc.objects.push(actor);
}

const renameById = {
  '37308997-579d-48a2-9b79-0dcc95890516': '我方单位_武夫',
  '3ab1360e-8c53-49f0-ba71-cfa35441f3f4': '敌方单位_疯子',
  '1d492370-97d6-41fc-9fbf-1c54d82c37ef': '敌方单位_污染兽'
};
for (const obj of doc.objects) {
  if (renameById[obj.id]) {
    obj.name = renameById[obj.id];
    obj.sortingLayer = '前景';
    obj.orderInLayer = 30;
  }
  if (obj.name === '背景') {
    obj.spritePath = '';
    obj.spriteColor = '#ff202822';
    obj.scale = { x: 14.8, y: 10.0 };
    obj.position = { x: 0, y: 0 };
    obj.sortingLayer = '背景';
    obj.orderInLayer = 0;
  }
}

upsert(baseActor('battle-arena-floor', '战斗场地底板', 'builtin/Sprite', 0, 0, 1480, 1000, '#ff202822', 1));
upsert(baseActor('battle-spawn-team-1', '出生区_team_1', 'builtin/Sprite', -430, 0, 260, 360, '#334d88cc', 5, '出生区'));
upsert(baseActor('battle-spawn-team-2', '出生区_team_2', 'builtin/Sprite', 430, 0, 260, 360, '#33884d4dcc', 5, '出生区'));
upsert(baseActor('battle-obstacle-center', '障碍_中央', 'builtin/Sprite', 0, 20, 120, 120, '#ff4a535d', 20, '障碍'));
upsert(baseActor('battle-range-preview', '释放范围预览', 'builtin/Sprite', -130, -10, 360, 120, '#665bd3ff', 18, '释放范围'));
upsert(baseActor('battle-target-highlight', '合法目标高亮', 'builtin/Sprite', 300, -60, 150, 150, '#66ffdf6e', 25, '目标高亮'));

fs.writeFileSync(path, JSON.stringify(doc, null, 2) + '\n');
NODE
```

Expected: command exits with code 0.
