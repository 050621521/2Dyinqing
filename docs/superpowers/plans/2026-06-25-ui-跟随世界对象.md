# UI 跟随世界对象（路线A）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给引擎加一个通用能力——运行时让某个 UI 实例「跟随」场景里的某个对象（Actor），UI 随对象在游戏视口里移动；首个用途是战斗地图界面里每个单位头顶的血条/蓝条。

**Architecture:** 在 `UIInstance` 上加「跟随目标 actorId + 偏移」字段；`UIRuntime` 提供设置/清除跟随的接口；`GameViewport::renderUI` 对带跟随目标的实例，按目标对象世界坐标投影到屏幕、平移到对象上方再绘制；新增蓝图节点 `UI.Follow`（中文「跟随对象」）让设计者在蓝图里绑定。先做小的 `头顶信息.ui`，再用既有无界面测试驱动 `BPRuntime`+`UIRuntime` 验证，最后用战斗关卡里的示例单位 + 一段关卡蓝图肉眼验证。

**Tech Stack:** C++17、Qt6 Widgets、CMake；无界面测试用既有 `launcher/uitest` 直接驱动 `BPRuntime`+`UIRuntime`。

```cpp
void UIRuntime::setFollowActor(const QString& instanceId, const QString& actorId,
                               float offsetX, float offsetY) {
    UIInstance* inst = findInstance(instanceId);
    if (!inst) return;
    inst->followActorId = actorId;
    inst->followOffsetX = offsetX;
    inst->followOffsetY = offsetY;
    emit uiStateChanged();
}

void UIRuntime::setFollowActorRef(const QString& ref, const QString& actorId,
                                  float offsetX, float offsetY) {
    UIInstance* inst = findInstance(ref);   // 先按实例 id
    if (!inst) inst = findByName(ref);        // 再按 UI 名
    if (!inst) return;
    setFollowActor(inst->instanceId, actorId, offsetX, offsetY);
}

void UIRuntime::clearFollow(const QString& instanceId) {
    if (UIInstance* inst = findInstance(instanceId)) {
        inst->followActorId.clear();
        emit uiStateChanged();
    }
}
```

```cpp
QPointF GameViewport::worldToScreen(QPointF world, const QRectF& camRect, const ActorData& cam) {
    const float halfH  = cam.cameraSize;
    const float aspect = cam.cameraResH > 0
                         ? (float)cam.cameraResW / cam.cameraResH
                         : 1.7778f;
    const float halfW  = halfH * aspect;
    const float scaleX = (float)camRect.width()  / (halfW * 2.0f);
    const float scaleY = (float)camRect.height() / (halfH * 2.0f);
    return QPointF(
        camRect.center().x() + (world.x() - cam.x) * scaleX,
        camRect.center().y() - (world.y() - cam.y) * scaleY
    );
}

QPointF GameViewport::cameraWorldToScreen(QPointF world, const QRectF& camRect,
                                           const ActorData& cam) const {
    return worldToScreen(world, camRect, cam);
}
```

```cpp
            p.translate(-canonicalW / 2.0f + inst->followOffsetX,
                        -canonicalH / 2.0f + inst->followOffsetY);
        } else {
            p.translate(camRect.left() + inst->screenX, camRect.top() + inst->screenY);
            p.scale(sx, sy);
        }
        const UIDocument& doc = inst->docCopy;
        for (const UIWidget& root : doc.rootWidgets())
            renderWidget(p, root, canonicalRect, doc);
        p.restore();
    }
```

```python
#!/usr/bin/env python3
import json, uuid, copy
PATH="/Users/kwy/Documents/2Dyinqing/MyGame/Levels/战斗地图界面.level"
d=json.load(open(PATH))
objs=d["objects"]
template=copy.deepcopy([o for o in objs if o["name"]=="背景"][0])

def set_pos(o, x, y):
    p=o.get("position")
    if isinstance(p, dict):       o["position"]={**p,"x":x,"y":y}
    elif isinstance(p, list):     o["position"]=[x,y]+list(p[2:])
    else:                          o["x"]=x; o["y"]=y

# 清掉上次生成的示例单位（名字以「示例单位_」开头）
objs[:] = [o for o in objs if not o.get("name","").startswith("示例单位_")]

units=[("示例单位_甲",-300,0),("示例单位_乙",0,80),("示例单位_丙",300,-60)]
unit_ids=[]
for name,x,y in units:
    o=copy.deepcopy(template)
    o["id"]=str(uuid.uuid4()); o["name"]=name
    o["bpClass"]="builtin/Sprite"
    o["tag"]="未标记"
    o["spriteColor"]="#ffffffff"   # 不染色：纯白
    o["spriteVisible"]=True
    set_pos(o,x,y)
    objs.append(o); unit_ids.append(o["id"])

bp=d["blueprint"]; bp.setdefault("nodes",[]); bp.setdefault("connections",[])
bp["nodes"][:]      = [n for n in bp["nodes"]       if not n["id"].startswith("demo_")]
bp["connections"][:]= [c for c in bp["connections"] if not c["id"].startswith("demo_")]

def node(nid,typ,params,x,y): return {"id":nid,"params":params,"type":typ,"x":x,"y":y}
def conn(fn,fp,tn,tp): return {"id":"demo_"+fn+"_"+tn,"fromNode":fn,"fromPin":fp,"toNode":tn,"toPin":tp}

bp["nodes"].append(node("demo_begin","Event.BeginPlay",{},-600,-300))
prev=("demo_begin","exec_out")
for i,uid in enumerate(unit_ids):
    cnode=f"demo_create{i}"; fnode=f"demo_follow{i}"
    bp["nodes"].append(node(cnode,"UI.Create",{"widgetRef":"头顶信息"}, -300, -300+i*180))
    bp["nodes"].append(node(fnode,"UI.Follow",{"actorId":uid,"offsetUp":"150"}, 0, -300+i*180))
    bp["connections"].append(conn(prev[0],prev[1],cnode,"exec_in"))
    bp["connections"].append(conn(cnode,"exec_out",fnode,"exec_in"))
    bp["connections"].append(conn(cnode,"widgetRef",fnode,"widgetRef"))
    prev=(fnode,"exec_out")

json.dump(d,open(PATH,"w",encoding="utf-8"),ensure_ascii=False,indent=4)
print("units",len(unit_ids),"nodes",len(bp['nodes']),"conns",len(bp['connections']))
```
